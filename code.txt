class RiskRegister(models.Model):

        ('PERSONNEL', 'Personnel'),
        ('TECHNICAL', 'Technical'),
        ('SCHEDULE', 'Schedule'),
        ('COST', 'Cost'),
        ('QUALITY', 'Quality'),
        ('EXTERNAL', 'External'),
        ('ORGANIZATIONAL', 'Organizational'),
        ('PROJECT_MANAGEMENT', 'Project Management'),
    ]
    
    SOURCE_CHOICES = [
        ('PLANNING', 'Planning'),
        ('REQUIREMENT', 'Requirement'),
        ('DESIGN', 'Design'),
        ('CODING', 'Coding and Unit Testing'),
        ('TESTING', 'System/Integration Testing'),
        ('DEPLOYMENT', 'Deployment'),
        ('MAINTENANCE', 'Maintenance'),
        ('COMPLIANCE', 'Compliance to Standard/s'),
    ]
    
    PHASE_CHOICES = [
        ('REQUIREMENT', 'Requirement'),
        ('PLANNING', 'Planning'),
        ('DESIGN', 'Design'),
        ('CODING', 'Coding'),
        ('UNIT_TESTING', 'Unit Testing'),
        ('SYSTEM_TESTING', 'System Testing'),
        ('INTEGRATION_TESTING', 'Integration Testing'),
        ('SPRINT_PLANNING', 'Sprint Planning'),
    ]

        ('ACCEPT', 'Accept'),
        ('AVOID', 'Avoid'),
        ('MITIGATE', 'Mitigate'),
        ('TRANSFER', 'Transfer'),
        ('ESCALATE', 'Escalate'),
    ]
    
    STATUS_CHOICES = [
        ('OPEN', 'Open'),
        ('IN_PROGRESS', 'In Progress'),
        ('MITIGATED', 'Mitigated'),
        ('CLOSED', 'Closed'),
        ('ESCALATED', 'Escalated'),
    ]
    
    risk_id = models.CharField(max_length=20, unique=True)
    description = models.TextField()
    originator_name = models.CharField(max_length=100)
    category = models.CharField(max_length=50, choices=CATEGORY_CHOICES)
    source = models.CharField(max_length=50, choices=SOURCE_CHOICES)
    identification_date = models.DateField()
    phase = models.CharField(max_length=50, choices=PHASE_CHOICES)
    treatment_option = models.CharField(max_length=20, choices=TREATMENT_OPTIONS)
    treatment_rationale = models.TextField(blank=True)
    effort_required = models.CharField(max_length=200, blank=True)
    treatment_schedule = models.TextField(blank=True)
    success_criteria = models.TextField(blank=True)
    cancellation_criteria = models.TextField(blank=True)
    monitoring_frequency = models.CharField(max_length=100, blank=True)
    threshold = models.CharField(max_length=100, blank=True)
    trigger = models.CharField(max_length=200, blank=True)
    probability = models.IntegerField(validators=[MinValueValidator(1), MaxValueValidator(5)])
    impact = models.IntegerField(validators=[MinValueValidator(1), MaxValueValidator(5)])
    risk_exposure = models.IntegerField(editable=False)  # Calculated field
    mitigation_plan = models.TextField()
    contingency_plan = models.TextField(blank=True)
    verification_methods = models.TextField(blank=True)
    stakeholders = models.TextField(blank=True)
    responsibility = models.CharField(max_length=100)
    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default='OPEN')
    remarks = models.TextField(blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    class Meta:
        ordering = ['risk_id']
        verbose_name = 'Risk Register'
        verbose_name_plural = 'Risk Registers'
    
    def save(self, *args, **kwargs):
        # Calculate risk exposure
        self.risk_exposure = self.probability * self.impact
        super().save(*args, **kwargs)
    
    def __str__(self):
        return f"{self.risk_id} - {self.description[:50]}"

from rest_framework import viewsets, status

class RiskRegisterViewSet(viewsets.ModelViewSet):
    queryset = RiskRegister.objects.all()
    serializer_class = RiskRegisterSerializer
    
    def get_queryset(self):
        queryset = RiskRegister.objects.all()
        
        # Search functionality
        search = self.request.query_params.get('search', None)
        if search:
            queryset = queryset.filter(
                Q(risk_id__icontains=search) |
                Q(description__icontains=search) |
                Q(originator_name__icontains=search) |
                Q(responsibility__icontains=search) |
                Q(mitigation_plan__icontains=search)
            )
        
        # Filter by category
        category = self.request.query_params.get('category', None)
        if category:
            queryset = queryset.filter(category=category)
        
        # Filter by status
        status_filter = self.request.query_params.get('status', None)
        if status_filter:
            queryset = queryset.filter(status=status_filter)
        
        # Filter by phase
        phase = self.request.query_params.get('phase', None)
        if phase:
            queryset = queryset.filter(phase=phase)
        
        # Filter by risk level
        risk_level = self.request.query_params.get('risk_level', None)
        if risk_level:
            if risk_level == 'HIGH':
                queryset = queryset.filter(risk_exposure__gte=12)
            elif risk_level == 'MEDIUM':
                queryset = queryset.filter(risk_exposure__gte=6, risk_exposure__lt=12)
            elif risk_level == 'LOW':
                queryset = queryset.filter(risk_exposure__lt=6)
        
        return queryset
    
    @action(detail=False, methods=['get'])
    def profile(self, request):
        """Get risk profile by phase"""
        phases = [
            'REQUIREMENT', 'PLANNING', 'DESIGN', 'CODING',
            'UNIT_TESTING', 'SYSTEM_TESTING', 'INTEGRATION_TESTING', 'SPRINT_PLANNING'
        ]
        
        total_risks = self.get_queryset().count()
        profile_data = []
        
        for phase in phases:
            count = self.get_queryset().filter(phase=phase).count()
            percentage = (count / total_risks * 100) if total_risks > 0 else 0
            profile_data.append({
                'phase': phase,
                'count': count,
                'percentage': round(percentage, 2)
            })
        
        return Response(profile_data)
    
    @action(detail=False, methods=['get'])
    def statistics(self, request):
        """Get risk statistics"""
        queryset = self.get_queryset()
        
        stats = {
            'total_risks': queryset.count(),
            'open_risks': queryset.filter(status='OPEN').count(),
            'high_risks': queryset.filter(risk_exposure__gte=12).count(),
            'medium_risks': queryset.filter(risk_exposure__gte=6, risk_exposure__lt=12).count(),
            'low_risks': queryset.filter(risk_exposure__lt=6).count(),
            'by_category': list(queryset.values('category').annotate(count=Count('id'))),
            'by_phase': list(queryset.values('phase').annotate(count=Count('id'))),
            'average_exposure': queryset.aggregate(Avg('risk_exposure'))['risk_exposure__avg'] or 0,
        }
        
        return Response(stats)
    
    @action(detail=False, methods=['post'])
    def import_excel(self, request):
        """Import risks from Excel file"""
        if 'file' not in request.FILES:
            return Response({'error': 'No file provided'}, status=status.HTTP_400_BAD_REQUEST)
        
        excel_file = request.FILES['file']
        handler = RiskOpportunityExcelHandler()
        
        try:
            data = handler.import_risks(excel_file)
            return Response({
                'message': 'Risks imported successfully',
                'imported': len(data)
            }, status=status.HTTP_201_CREATED)
        except Exception as e:
            return Response({'error': str(e)}, status=status.HTTP_400_BAD_REQUEST)
    
    @action(detail=False, methods=['get'])
    def export_excel(self, request):
        """Export risks to Excel file"""
        handler = RiskOpportunityExcelHandler()
        excel_file = handler.export_risks(self.get_queryset())
        
        response = HttpResponse(
            excel_file,
            content_type='application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
        )
        response['Content-Disposition'] = 'attachment; filename=risk_register.xlsx'
        return response

    @action(detail=False, methods=['get'])
    def calculate_scores(self, request):
        """Calculate risk scores by QCD category"""
        queryset = self.get_queryset()
        
        scores = {
            'quality_score': queryset.filter(qcd_category='QUALITY').aggregate(Sum('risk_score'))['risk_score__sum'] or 0,
            'cost_score': queryset.filter(qcd_category='COST').aggregate(Sum('risk_score'))['risk_score__sum'] or 0,
            'delivery_score': queryset.filter(qcd_category='DELIVERY').aggregate(Sum('risk_score'))['risk_score__sum'] or 0,
        }
        scores['total_score'] = sum(scores.values())
        
        return Response(scores)

import pandas as pd
import openpyxl
from io import BytesIO
from datetime import datetime
from .models import RiskRegister, OpportunityRegister

class RiskOpportunityExcelHandler:
    def import_risks(self, excel_file):
        """Import risks from Excel file"""
        df = pd.read_excel(excel_file, sheet_name='Risk_Register', skiprows=1)
        
        imported_data = []
        
        for index, row in df.iterrows():
            if pd.isna(row.get('Risk Id.')):
                continue
            
            # Parse date
            date_str = row.get('Date of Identification\n(YYYY-MM-DD)', '')
            if isinstance(date_str, str):
                try:
                    date = datetime.strptime(date_str, '%Y-%m-%d').date()
                except:
                    continue
            else:
                date = date_str
            
            # Create or update risk
            risk, created = RiskRegister.objects.update_or_create(
                risk_id=str(row.get('Risk Id.', '')),
                defaults={
                    'description': row.get('Risk Description', ''),
                    'originator_name': row.get('Risk Originator Name', ''),
                    'category': self._map_category(row.get('Risk Category', '')),
                    'source': self._map_source(row.get('Risk Source', '')),
                    'identification_date': date,
                    'phase': self._map_phase(row.get('Phase of Identification', '')),
                    'treatment_option': self._map_treatment(row.get('Risk Treatment Option', '')),
                    'treatment_rationale': row.get('Rationale to choose Risk Treatment Option', ''),
                    'effort_required': row.get('Effort Required for Risk Treatment', ''),
                    'treatment_schedule': row.get('Risk Treatment Schedule', ''),
                    'success_criteria': row.get('Success criteria for Risk Treatment activities', ''),
                    'cancellation_criteria': row.get('Criteria for cancellation of Risk Treatment Activities', ''),
                    'monitoring_frequency': row.get('Frequency of monitoring Risk treatment activities.', ''),
                    'threshold': row.get('Threshold', ''),
                    'trigger': row.get('Trigger', ''),
                    'probability': int(row.get('Probability', 1)),
                    'impact': int(row.get('Impact', 1)),
                    'mitigation_plan': row.get('Mitigation Plan', ''),
                    'contingency_plan': row.get('Contingency Plan', ''),
                    'verification_methods': row.get('Verification methods for Mitigation/Contingency Plan', ''),
                    'stakeholders': row.get('List of stakeholders', ''),
                    'responsibility': row.get('Responsibility \n(Assigned to)', ''),
                    'status': self._map_status(row.get('Status*', '')),
                    'remarks': row.get('Remarks', ''),
                }
            )
            
            imported_data.append(risk)
        
        return imported_data
    
    def import_opportunities(self, excel_file):
        """Import opportunities from Excel file"""
        df = pd.read_excel(excel_file, sheet_name='Opportunity_Register', skiprows=1)
        
        imported_data = []
        
        for index, row in df.iterrows():
            if pd.isna(row.get('Opportunity Id.')):
                continue
            
            # Parse date
            date_str = row.get('Date of Identification\n(YYYY-MM-DD)', '')
            if isinstance(date_str, str):
                try:
                    date = datetime.strptime(date_str, '%Y-%m-%d').date()
                except:
                    continue
            else:
                date = date_str
            
            # Create or update opportunity
            opportunity, created = OpportunityRegister.objects.update_or_create(
                opportunity_id=str(row.get('Opportunity Id.', '')),
                defaults={
                    'description': row.get('Opportunity Description', ''),
                    'category': self._map_opp_category(row.get('Opportunity Category', '')),
                    'source': row.get('Opportunity Source', ''),
                    'identification_date': date,
                    'phase': self._map_phase(row.get('Phase of Identification', '')),
                    'cost': int(row.get('Cost\n[High(1), Medium(2), Low(3)]', 2)),
                    'benefit': int(row.get('Benefit\n[High(3), Medium(2), Low(1)]', 2)),
                    'leverage_plan': row.get('Leverage Plan to maximize Opportunities identified', ''),
                    'responsibility': row.get('Responsibility \n(Assigned to)', ''),
                    'status': self._map_opp_status(row.get('Status*', '')),
                    'remarks': row.get('Remarks', ''),
                }
            )
            
            imported_data.append(opportunity)
        
        return imported_data
    
    def export_risks(self, queryset):
        """Export risks to Excel file"""
        wb = openpyxl.Workbook()
        ws = wb.active
        ws.title = 'Risk_Register'
        
        # Add header row
        headers = [
            'Risk Id.', 'Risk Description', 'Risk Originator Name', 'Risk Category',
            'Risk Source', 'Date of Identification\n(YYYY-MM-DD)', 'Phase of Identification',
            'Risk Treatment Option', 'Rationale to choose Risk Treatment Option',
            'Effort Required for Risk Treatment', 'Risk Treatment Schedule',
            'Success criteria for Risk Treatment activities',
            'Criteria for cancellation of Risk Treatment Activities',
            'Frequency of monitoring Risk treatment activities.',
            'Threshold', 'Trigger', 'Probability', 'Impact', 'Risk Exposure\n(Probability x Impact)',
            'Mitigation Plan', 'Contingency Plan',
            'Verification methods for Mitigation/Contingency Plan',
            'List of stakeholders', 'Responsibility \n(Assigned to)', 'Status*', 'Remarks'
        ]
        
        # Style headers
        for col, header in enumerate(headers, 1):
            cell = ws.cell(row=1, column=col, value=header)
            cell.font = openpyxl.styles.Font(bold=True)
            cell.fill = openpyxl.styles.PatternFill(start_color="00CED1", end_color="00CED1", fill_type="solid")
            cell.alignment = openpyxl.styles.Alignment(horizontal='center', vertical='center', wrap_text=True)
        
        # Add data
        for row_num, risk in enumerate(queryset, 2):
            ws.cell(row=row_num, column=1, value=risk.risk_id)
